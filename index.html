function doPost(e) { try { var data = JSON.parse(e.postData.contents); var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); // Fotoğrafların yükleneceği Drive klasörünü belirleyin/oluşturun var folderName = "Öğrenci Ödev Fotoğrafları"; var folders = DriveApp.getFoldersByName(folderName); var folder; if (folders.hasNext()) { folder = folders.next(); } else { folder = DriveApp.createFolder(folderName); } var fileUrls = []; // Fotoğrafları Drive'a kaydetme if (data.sayfaFotograflari && data.sayfaFotograflari.length > 0) { data.sayfaFotograflari.forEach(function(fileObj, index) { var base64Data = fileObj.data.split(",")[1]; var decodedData = Utilities.base64Decode(base64Data); var blob = Utilities.newBlob(decodedData, fileObj.type, (data.ogrenciAdi || "Ogrenci") + "_" + (index + 1) + "_" + fileObj.name); var file = folder.createFile(blob); file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); fileUrls.push(file.getUrl()); }); } // Tabloya yeni satır ekleme sheet.appendRow([ new Date(), data.ogrenciAdi || "", (data.ders || []).join(", "), data.duyguDurumu || "", data.konuNotlari || "", fileUrls.join("\n") ]); return ContentService.createTextOutput(JSON.stringify({ "result": "success" })) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({ "result": "error", "error": err.toString() })) .setMimeType(ContentService.MimeType.JSON); } }
